const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition; const speakButton = document.querySelector("#speakButton"); const speakButtonText = document.querySelector("#speakButtonText"); const hearButton = document.querySelector("#hearButton"); const analyzeButton = document.querySelector("#analyzeButton"); const transcriptInput = document.querySelector("#manualTranscript"); const locationInput = document.querySelector("#locationInput"); const speechModel = document.querySelector("#speechModel"); const languageModel = document.querySelector("#languageModel"); const statusText = document.querySelector("#statusText"); const statusDot = document.querySelector("#statusDot"); const languageValue = document.querySelector("#languageValue"); const symptomsValue = document.querySelector("#symptomsValue"); const specialtyValue = document.querySelector("#specialtyValue"); const modelValue = document.querySelector("#modelValue"); const urgencyBlock = document.querySelector("#urgencyBlock"); const urgencyLabel = document.querySelector("#urgencyLabel"); const urgencyMessage = document.querySelector("#urgencyMessage"); const disclaimerText = document.querySelector("#disclaimerText"); const providerList = document.querySelector("#providerList"); const sourceList = document.querySelector("#sourceList"); const careSearch = document.querySelector("#careSearch"); const searchInput = document.querySelector("#searchInput"); let latestSpokenResponse = ""; let recognition; let isListening = false; function setStatus(text, listening = false) { statusText.textContent = text; statusDot.classList.toggle("listening", listening); } function providerCard(provider) { const consultTypes = provider.consultation_types.map((type) => `${type}`).join(""); const professions = provider.professions.slice(0, 4).map((profession) => `${profession}`).join(""); return `

${provider.name}

${provider.business_address}

${provider.ownership} ${provider.provider_type} ${provider.specialty} ${provider.distance_km.toFixed(1)} km
${professions}
${consultTypes}

${provider.availability}

${provider.rating.toFixed(1)} relevance score

${provider.source_name}
`; } function renderProviders(providers) { if (!providers.length) { providerList.innerHTML = `

No providers matched this search yet.

`; return; } providerList.innerHTML = providers.map(providerCard).join(""); } function renderAnalysis(data) { languageValue.textContent = data.detected_language; symptomsValue.textContent = data.symptoms.length ? data.symptoms.map((symptom) => symptom.name).join(", ") : "No clear symptom detected"; specialtyValue.textContent = data.suggested_specialty; modelValue.textContent = `${data.model_stack.speech_to_text} + ${data.model_stack.language_understanding}`; urgencyBlock.dataset.level = data.assessment.urgency; urgencyLabel.textContent = data.assessment.label; urgencyMessage.textContent = `${data.assessment.message} ${data.recommended_next_action}`; disclaimerText.textContent = data.disclaimer; latestSpokenResponse = data.spoken_response; renderProviders(data.providers); } async function analyzeTranscript() { const transcript = transcriptInput.value.trim(); if (!transcript) { setStatus("Add symptoms first, then analyze"); transcriptInput.focus(); return; } setStatus("Understanding symptoms..."); const response = await fetch("/api/voice/analyze", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ transcript, location: locationInput.value.trim() || null, speech_model: speechModel.value, language_model: languageModel.value, }), }); if (!response.ok) { setStatus("Analysis failed. Please try again."); return; } const data = await response.json(); renderAnalysis(data); setStatus("Analysis ready"); } function startListening() { if (!SpeechRecognition) { setStatus("Chrome speech recognition is unavailable. Type symptoms instead."); return; } recognition = new SpeechRecognition(); recognition.lang = "en-IN"; recognition.interimResults = true; recognition.continuous = false; recognition.onstart = () => { isListening = true; speakButtonText.textContent = "Listening..."; setStatus("Listening in Chrome", true); }; recognition.onresult = (event) => { const transcript = Array.from(event.results) .map((result) => result[0].transcript) .join(" "); transcriptInput.value = transcript; }; recognition.onerror = (event) => { setStatus(`Voice error: ${event.error}`); }; recognition.onend = () => { isListening = false; speakButtonText.textContent = "Hold to speak"; setStatus("Voice captured"); if (transcriptInput.value.trim()) { analyzeTranscript(); } }; recognition.start(); } function stopListening() { if (recognition && isListening) { recognition.stop(); } } function speakLatestResult() { if (!latestSpokenResponse) { setStatus("Analyze symptoms first, then hear the result"); return; } window.speechSynthesis.cancel(); window.speechSynthesis.speak(new SpeechSynthesisUtterance(latestSpokenResponse)); } async function loadProviders() { const response = await fetch("/api/providers"); if (response.ok) { renderProviders(await response.json()); } } async function loadSources() { const response = await fetch("/api/data-sources"); if (!response.ok) return; const sources = await response.json(); sourceList.innerHTML = sources .map( (source) => ` ${source.name} ${source.use_case} `, ) .join(""); } async function searchCare(event) { event.preventDefault(); const query = searchInput.value.trim(); if (!query) return; const params = new URLSearchParams({ query }); if (locationInput.value.trim()) params.set("location", locationInput.value.trim()); const response = await fetch(`/api/search?${params.toString()}`); if (response.ok) { const data = await response.json(); renderProviders(data.results); setStatus("Search results updated"); } } speakButton.addEventListener("mousedown", startListening); speakButton.addEventListener("mouseup", stopListening); speakButton.addEventListener("mouseleave", stopListening); speakButton.addEventListener("touchstart", (event) => { event.preventDefault(); startListening(); }); speakButton.addEventListener("touchend", stopListening); analyzeButton.addEventListener("click", analyzeTranscript); hearButton.addEventListener("click", speakLatestResult); careSearch.addEventListener("submit", searchCare); loadProviders(); loadSources();
